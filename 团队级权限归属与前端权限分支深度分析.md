```
┌─────────────────────────────────────────────────────────────────────────────────────┐
│                                    数据模型关系                                        │
└─────────────────────────────────────────────────────────────────────────────────────┘

                          ┌──────────────────┐
                          │      teams       │

           │                       │                         │
           │ has_many              │ has_many                │
           │ :guest_memberships    │ :guest_memberships     │
           ▼                       ▼                         │
    ┌──────────────────┐      ┌──────────────────┐          │
    │ guest_memberships│◄─────│ guest_memberships│◄─────────┘
    │ (通过 Site)      │      │ (通过 Membership) │
    ├──────────────────┤      ├──────────────────┤
    │ id (PK)          │      │ id (PK)          │
    │ site_id (FK)     │      │ team_membership_ │
    │ role (Enum)      │      │ id (FK)          │
    │                  │      │ site_id (FK)     │
    │                  │      │ role (Enum)      │
    │                  │      │ (:viewer/:editor)│
    └──────────────────┘      └──────────────────┘

关键说明：
1. ownerships 是 team_memberships 的筛选视图（where: [role: :owner]）
2. guest_memberships 通过两条路径关联：
   - 直接关联到 site（用于查询特定站点的访客）
   - 关联到 team_membership（用于查询特定用户的所有访客权限）
3. 一个站点的所有者 = 所属团队的所有者（通过 team 间接关联）
```

#### 1.2.4 GuestMembership 模型

**文件位置**：`lib/plausible/teams/guest_membership.ex:12-27`

```elixir
schema "guest_memberships" do
  # 访客角色仅限于这两种（不能是 admin/owner）
  field :role, Ecto.Enum, values: [:viewer, :editor]
  
  # 双重关联
  belongs_to :team_membership, Plausible.Teams.Membership  # 关联到 team_memberships 表
  belongs_to :site, Plausible.Site                         # 关联到具体站点
  
  timestamps()
end
```

**关键设计**：

1. **双重关联目的**：
   - 通过 `team_membership` 可以查询："用户 A 在哪些站点有访客权限？"
   - 通过 `site` 可以查询："站点 B 有哪些访客成员？"

2. **角色限制**：访客只能是 `:viewer` 或 `:editor`，不能拥有更高权限

### 1.3 权限归属查询逻辑

#### 1.3.1 站点角色查询（核心函数）

**文件位置**：`lib/plausible/teams/memberships.ex:80-101`

```elixir
@spec site_role(Plausible.Site.t(), Auth.User.t() | nil) ::
        {:ok, {:team_member | :guest_member, Teams.Membership.role()}} | {:error, :not_a_member}

def site_role(site, user) do
  result =
    from(u in Auth.User,
      inner_join: tm in assoc(u, :team_memberships),
      left_join: gm in assoc(tm, :guest_memberships),
      where: tm.team_id == ^site.team_id and tm.user_id == ^user.id,
      # 关键条件：非访客角色直接通过；访客角色需要有对应站点的 guest_membership
      where: tm.role != :guest or gm.site_id == ^site.id,
      select: {tm.role, gm.role}
    )
    |> Repo.one()

  case result do
    # 如果 team_membership.role 是 :guest，则使用 guest_membership.role
    {:guest, role} -> {:ok, {:guest_member, role}}
    # 否则直接使用 team_membership.role
    {role, _} -> {:ok, {:team_member, role}}
    _ -> {:error, :not_a_member}
  end
end
```

#### 1.3.2 当前团队和角色设置（AuthPlug）

**文件位置**：`lib/plausible_web/plugs/auth_plug.ex:18-93`

```elixir
def call(conn, _opts) do
  case UserAuth.get_user_session(conn) do
    {:ok, user_session} ->
      user = user_session.user
      
      # 从多个来源获取当前团队 ID
      current_team_id_from_session = Plug.Conn.get_session(conn, "current_team_id")
      
      current_team_id =
        conn.params["__team"] ||          # 1. URL 参数（如 ?__team=uuid）
        current_team_id_from_session ||   # 2. Session 存储
        user.last_team_identifier          # 3. 用户最后使用的团队
      
      # 根据 team_id 查找对应的成员关系和角色
      {current_team, current_team_role} =
        if current_team_id do
          team_membership =
            Enum.find(user.team_memberships, %{}, 
              &(&1.team.identifier == current_team_id))
          
          {Map.get(team_membership, :team), Map.get(team_membership, :role)}
        else
          {nil, nil}
        end
      
      # 处理 my_team（未完成设置的团队，用户是 owner）
      my_team =
        user.team_memberships
        |> Enum.find(%{}, &(&1.role == :owner and &1.team.setup_complete == false))
        |> Map.get(:team)
      
      # 设置 conn.assigns
      conn
      |> assign(:current_user, user)
      |> assign(:current_user_session, user_session)
      |> assign(:my_team, my_team)
      # 如果没有当前团队，使用 my_team（未完成设置的个人团队）
      |> assign(:current_team, current_team || my_team)
      # 如果使用 my_team，角色是 :owner
      |> assign(:current_team_role, current_team_role || (my_team && :owner))
      |> assign(:teams_count, length(user.team_memberships))
      # 排序：owner 在前，按 team_id 排序
      |> assign(:teams, sorted_teams_list)
      |> assign(:more_teams?, teams_count > 3)
      
    {:error, :session_expired, user_session} ->
      assign(conn, :expired_session, user_session)
      
    _ ->
      conn
  end
end
```

**当前团队选择优先级**：

| 优先级 | 来源 | 说明 |
|--------|------|------|
| 1 | URL 参数 `__team` | 如 `/sites?__team=uuid-value` |
| 2 | Session `current_team_id` | 用户上次选择的团队 |
| 3 | `user.last_team_identifier` | 数据库中存储的最后使用团队 |
| 4 | `my_team` | 未完成设置的个人团队（用户是 owner） |

#### 1.3.3 LiveView 中的 AuthContext

**文件位置**：`lib/plausible_web/live/auth_context.ex:19-103`

```elixir
def on_mount(:default, _params, session, socket) do
  socket =
    socket
    # 1. 获取用户会话和用户
    |> assign_new(:current_user_session, fn -> ... end)
    |> assign_new(:current_user, fn context -> ... end)
    
    # 2. 从 session 中获取团队
    |> assign_new(:team_from_session, fn
      %{current_user: nil} -> nil
      %{current_user: user} ->
        if current_team_id = session["current_team_id"] do
          user.team_memberships
          |> Enum.find(%{}, &(&1.team.identifier == current_team_id))
          |> Map.get(:team)
        end
    end)
    
    # 3. 获取 my_team（未完成设置的个人团队）
    |> assign_new(:my_team, fn
      %{current_user: nil} -> nil
      %{current_user: user} ->
        user.team_memberships
        |> Enum.find(%{}, &(&1.role == :owner and &1.team.setup_complete == false))
        |> Map.get(:team)
    end)
    
    # 4. 确定当前团队（优先级同 AuthPlug）
    |> assign_new(:current_team, fn
      %{current_user: nil} -> nil
      %{team_from_session: %{} = team_from_session} -> team_from_session
      %{my_team: %{} = my_team} -> my_team
      _ -> nil
    end)
    
    # 5. 确定当前团队角色
    |> assign_new(:current_team_role, fn
      %{current_user: user = %{}, current_team: current_team = %{}} ->
        # 在 user.team_memberships 中查找对应 team 的角色
        Enum.find_value(user.team_memberships, fn team_membership ->
          if team_membership.team_id == current_team.id do
            team_membership.role
          end
        end)
      %{my_team: %{}} -> :owner  # my_team 的角色一定是 owner
      _ -> nil
    end)
    
    # 6. 其他团队相关信息
    |> assign_new(:teams, fn ... end)
    |> assign_new(:teams_count, fn ... end)
    |> assign_new(:more_teams?, fn ... end)

  {:cont, socket}
end
```

**关键同步要求**：

> 注释：`Must be kept in sync with PlausibleWeb.AuthPlug`

这意味着 `Live.AuthContext` 和 `AuthPlug` 必须保持逻辑一致，确保在普通 HTTP 请求和 LiveView 连接中，用户的 `current_team` 和 `current_team_role` 是相同的。

### 2.2 权限分支逻辑（真实代码）

#### 2.2.1 片段（Segment）权限控制

**文件位置**：`assets/js/dashboard/filtering/segments.ts:13-267`

```typescript
// 与后端同步的角色数组
/** keep in sync with Plausible.Segments */
const ROLES_WITH_MAYBE_SITE_SEGMENTS = [Role.admin, Role.editor, Role.owner]
const ROLES_WITH_PERSONAL_SEGMENTS = [
  Role.billing,
  Role.viewer,
  Role.admin,
  Role.editor,
  Role.owner
]
```

**权限矩阵**：

| 功能 | 角色要求 | 说明 |
|------|----------|------|
| 查看"保存为片段"按钮 | `ROLES_WITH_MAYBE_SITE_SEGMENTS` 或 `ROLES_WITH_PERSONAL_SEGMENTS` | 几乎所有登录用户都能看到 |
| 创建/编辑站点级片段 | `ROLES_WITH_MAYBE_SITE_SEGMENTS` | admin/editor/owner |
| 创建/编辑个人片段 | `ROLES_WITH_PERSONAL_SEGMENTS` + 自己是所有者 | 几乎所有登录用户，但只能编辑自己的 |
| 列出站点级片段 | 站点功能可用即可 | 所有登录用户 |
| 列岀个人片段 | 自己是所有者 | 只能看到自己的 |

**真实代码 - 检查是否能展开/编辑片段**：

```typescript
// assets/js/dashboard/filtering/segments.ts:198-223
export function canExpandSegment({
  segment,
  user
}: {
  segment: Pick<SavedSegment, 'id' | 'owner_id' | 'type'>
  user: UserContextValue
}) {
  // 站点级片段：admin/editor/owner 可以编辑
  if (
    segment.type === SegmentType.site &&
    user.loggedIn &&
    ROLES_WITH_MAYBE_SITE_SEGMENTS.includes(user.role)
  ) {
    return true
  }

  // 个人片段：只能编辑自己的，且角色在允许列表中
  if (
    segment.type === SegmentType.personal &&
    user.loggedIn &&
    ROLES_WITH_PERSONAL_SEGMENTS.includes(user.role) &&
    user.id === segment.owner_id  // 关键：必须是自己创建的
```

```typescript
// assets/js/dashboard/filtering/segments.ts:225-248
export function isListableSegment({
  segment,
  site,
  user
}: {
  segment: Pick<SavedSegment, 'id' | 'type' | 'owner_id'> | Pick<SavedSegmentPublic, ...>
  site: Pick<PlausibleSite, 'siteSegmentsAvailable'>
  user: UserContextValue
}) {
  // 站点级片段：只要站点功能可用就可见
  if (segment.type === SegmentType.site && site.siteSegmentsAvailable) {
    return true

  if (segment.type === SegmentType.personal) {

      return false
    }
    return segment.owner_id === user.id
  }

  return false
}
```

## 三、权限校验的双重保障机制

### 3.1 前端展示层控制

**目的**：优化用户体验，避免用户点击无权限的操作

**实现方式**：
1. **条件渲染** (`:if={@site_role in [:owner, :admin]}`)
2. **禁用状态** (`disabled={@site_role not in [:owner, :admin]}`)
3. **隐藏按钮** (不渲染 DOM 元素)
4. **不同提示信息** (根据角色显示不同的引导文案)

**示例**：

```elixir
def update_role(site, user_id, new_role_str, current_user, current_user_role) do
  new_role = String.to_existing_atom(new_role_str)
  
  case get_guest_membership(site.id, user_id) do
    {:ok, guest_membership} ->
      can_grant_role? =
        if guest_membership.team_membership.user_id == current_user.id do
          false  # 不能修改自己的角色
        else
          # 检查角色授予权限
          can_grant_role_to_other?(current_user_role, new_role)
        end
      
      if can_grant_role? do
        # 更新角色
        guest_membership
        |> Ecto.Changeset.change(role: new_role)
        |> Repo.update!()
        {:ok, guest_membership}
      else
        {:error, :not_allowed}
      end
    
    {:error, _} ->
      {:error, :no_guest}
  end
end

# 角色授予规则
defp can_grant_role_to_other?(:owner, :editor), do: true
defp can_grant_role_to_other?(:owner, :viewer), do: true
defp can_grant_role_to_other?(:admin, :editor), do: true
defp can_grant_role_to_other?(:admin, :viewer), do: true
defp can_grant_role_to_other?(_, _), do: false  # 其他情况都不允许
```
